cmake_minimum_required(VERSION 3.16)
project(pcmn_cpu LANGUAGES CXX)
add_library(pcmn_cpu SHARED affine_scan.cpp)
target_compile_features(pcmn_cpu PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(pcmn_cpu PRIVATE /O2 /arch:AVX2)
else()
    target_compile_options(pcmn_cpu PRIVATE -O3)
endif()
find_package(OpenMP)
if(OpenMP_CXX_FOUND)
    target_link_libraries(pcmn_cpu PRIVATE OpenMP::OpenMP_CXX)
endif()
